cmake_minimum_required(VERSION 3.15)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/include/PianoRollCore/Core.h" _piano_core_version_line
  REGEX "^inline constexpr const char\\* version = ")
string(REGEX MATCH "[0-9]+\\.[0-9]+\\.[0-9]+" _piano_core_version "${_piano_core_version_line}")
project(PianoRollCore VERSION ${_piano_core_version} LANGUAGES CXX)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/include/PianoRollCore/Core.h")

if(TARGET PianoRollCore::PianoRollCore)
  return()
endif()

add_library(PianoRollCore STATIC src/PianoRollCore.cpp src/Navigation.cpp src/Interaction.cpp src/Controllers.cpp src/Meter.cpp src/Playback.cpp
  src/Commands.cpp src/MusicTools.cpp src/NoteActions.cpp src/AutomationActions.cpp)
add_library(PianoRollCore::PianoRollCore ALIAS PianoRollCore)
target_include_directories(PianoRollCore PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_compile_features(PianoRollCore PUBLIC cxx_std_17)
set_target_properties(PianoRollCore PROPERTIES POSITION_INDEPENDENT_CODE ON)

set(_piano_core_has_juce OFF)
if(TARGET juce::juce_gui_extra)
  set(_piano_core_has_juce ON)
endif()
option(PIANO_ROLL_CORE_BUILD_JUCE_UI "Build shared JUCE command/settings and transform UI" ${_piano_core_has_juce})
if(PIANO_ROLL_CORE_BUILD_JUCE_UI)
  if(NOT TARGET juce::juce_gui_extra)
    message(FATAL_ERROR "Add JUCE before PianoRollCore when enabling the optional command UI")
  endif()
  add_library(PianoRollCommandUI STATIC juce/CommandPanel.cpp juce/CommandPreferences.cpp juce/TransformOverlay.cpp)
  add_library(PianoRollCore::CommandUI ALIAS PianoRollCommandUI)
  target_include_directories(PianoRollCommandUI PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/juce/include>)
  target_compile_features(PianoRollCommandUI PUBLIC cxx_std_17)
  target_compile_definitions(PianoRollCommandUI PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
  target_link_libraries(PianoRollCommandUI PUBLIC PianoRollCore::PianoRollCore juce::juce_gui_extra)
  set_target_properties(PianoRollCommandUI PROPERTIES POSITION_INDEPENDENT_CODE ON)
endif()

set(_piano_core_top_level OFF)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(_piano_core_top_level ON)
endif()
option(PIANO_ROLL_CORE_BUILD_TESTS "Build the focused PianoRollCore tests" ${_piano_core_top_level})
option(PIANO_ROLL_CORE_BUILD_EXAMPLE "Build the standalone adapter example" ${_piano_core_top_level})
option(PIANO_ROLL_CORE_BUILD_LAB "Build the standalone PianoRollLab interaction adapter" OFF)

if(PIANO_ROLL_CORE_BUILD_TESTS)
  enable_testing()
  add_executable(PianoRollCoreTests tests/CoreTests.cpp)
  target_link_libraries(PianoRollCoreTests PRIVATE PianoRollCore::PianoRollCore)
  add_test(NAME PianoRollCoreTests COMMAND PianoRollCoreTests)
  add_executable(PianoRollExperienceTests tests/ExperienceTests.cpp)
  target_link_libraries(PianoRollExperienceTests PRIVATE PianoRollCore::PianoRollCore)
  add_test(NAME PianoRollExperienceTests COMMAND PianoRollExperienceTests)
  find_package(Threads REQUIRED)
  add_executable(PianoRollPlaybackTests tests/PlaybackTests.cpp)
  target_link_libraries(PianoRollPlaybackTests PRIVATE PianoRollCore::PianoRollCore Threads::Threads)
  add_test(NAME PianoRollPlaybackTests COMMAND PianoRollPlaybackTests)
  add_executable(PianoRollCommandTests tests/CommandTests.cpp)
  target_link_libraries(PianoRollCommandTests PRIVATE PianoRollCore::PianoRollCore)
  add_test(NAME PianoRollCommandTests COMMAND PianoRollCommandTests)
  add_executable(PianoRollNoteActionTests tests/NoteActionTests.cpp)
  target_link_libraries(PianoRollNoteActionTests PRIVATE PianoRollCore::PianoRollCore)
  add_test(NAME PianoRollNoteActionTests COMMAND PianoRollNoteActionTests)
  add_executable(PianoRollAutomationActionTests tests/AutomationActionTests.cpp)
  target_link_libraries(PianoRollAutomationActionTests PRIVATE PianoRollCore::PianoRollCore)
  add_test(NAME PianoRollAutomationActionTests COMMAND PianoRollAutomationActionTests)
  add_executable(PianoRollLabTests tests/PianoRollLabTests.cpp)
  target_link_libraries(PianoRollLabTests PRIVATE PianoRollCore::PianoRollCore)
  target_include_directories(PianoRollLabTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/examples)
  add_test(NAME PianoRollLabTests COMMAND PianoRollLabTests)
endif()
if(PIANO_ROLL_CORE_BUILD_EXAMPLE)
  add_executable(PianoRollCoreExample examples/MinimalAdapter.cpp)
  target_link_libraries(PianoRollCoreExample PRIVATE PianoRollCore::PianoRollCore)
  add_executable(PianoRollPlaybackExample examples/PlaybackExample.cpp)
  target_link_libraries(PianoRollPlaybackExample PRIVATE PianoRollCore::PianoRollCore)
endif()

if(PIANO_ROLL_CORE_BUILD_LAB)
  if(NOT WIN32)
    message(FATAL_ERROR "PianoRollLab is currently implemented for the Win32 desktop fixture")
  endif()
  add_executable(PianoRollLab WIN32 examples/PianoRollLabWin32.cpp)
  target_include_directories(PianoRollLab PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/examples)
  target_link_libraries(PianoRollLab PRIVATE PianoRollCore::PianoRollCore user32 gdi32)
endif()
