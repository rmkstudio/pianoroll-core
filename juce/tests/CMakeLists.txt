cmake_minimum_required(VERSION 3.22)
project(PianoRollCommandUIChecks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(JUCE_SOURCE_DIR "" CACHE PATH "JUCE 7 source directory")
add_subdirectory("${JUCE_SOURCE_DIR}" juce EXCLUDE_FROM_ALL)
set(PIANO_ROLL_CORE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PIANO_ROLL_CORE_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. piano-roll-core EXCLUDE_FROM_ALL)
get_directory_property(_piano_core_version DIRECTORY ../.. DEFINITION PROJECT_VERSION)
juce_add_console_app(PianoRollCommandUIChecks VERSION "${_piano_core_version}"
  PRODUCT_NAME "Piano roll command UI checks")
target_sources(PianoRollCommandUIChecks PRIVATE CommandUITests.cpp)
target_compile_definitions(PianoRollCommandUIChecks PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1)
target_link_libraries(PianoRollCommandUIChecks PRIVATE PianoRollCore::CommandUI)
enable_testing()
add_test(NAME PianoRollCommandUIChecks COMMAND PianoRollCommandUIChecks "${CMAKE_CURRENT_BINARY_DIR}/test-output")
juce_add_console_app(PianoRollViewChecks VERSION "${_piano_core_version}"
  PRODUCT_NAME "Piano roll view checks")
target_sources(PianoRollViewChecks PRIVATE PianoRollViewTests.cpp)
target_link_libraries(PianoRollViewChecks PRIVATE PianoRollCore::View)
# Button::triggerClick is asynchronous; the checks pump the message loop.
target_compile_definitions(PianoRollViewChecks PRIVATE JUCE_MODAL_LOOPS_PERMITTED=1)
add_test(NAME PianoRollViewChecks COMMAND PianoRollViewChecks)
